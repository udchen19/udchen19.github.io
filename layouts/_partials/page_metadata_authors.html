{{/* Display author list. Copy of Hugo Blox Kit's template (modules/blox, 61f41d3667f1) with two changes that keep the author line the site had under the old Blox: the owner's name (is_owner in data/authors/) is bold, and a name links only when its author page is published. Upstream turns the bold off and links every name, which gives empty links while author pages are not rendered (content/authors/_index.md). Author notes (equal contribution) print as † with one key line, instead of Blox's tooltip, which needs a script this site does not load. */}} {{- $authors := .Params.authors -}} {{ if $authors }} {{- if not (reflect.IsSlice $authors) -}} {{- $authors = slice $authors -}} {{- end -}} {{ range $index, $raw := $authors }} {{- $slug := $raw | urlize -}} {{- $profile := partial "functions/get_author_profile" (dict "slug" $slug "title" $raw) -}} {{- $termPage := site.GetPage (printf "/authors/%s" $slug) -}} {{- $display := $profile.title | default $raw -}} {{- $highlight_name := $profile.is_owner -}} {{- if gt $index 0 }}, {{ end -}} {{- if and $termPage $termPage.RelPermalink -}} {{ $display }} {{- else -}} {{ $display }} {{- end -}} {{- /* Author notes print as a superscript † (no script), explained once after the list. */}} {{- $notes := $.Params.author_notes | default slice -}} {{- if and (lt $index (len $notes)) (index $notes $index) }}†{{ end -}} {{- end -}} {{- with uniq (.Params.author_notes | default slice) }}{{ with (delimit . "; ") }}† {{ . }}{{ end }}{{ end -}} {{- end -}}
